ACTIONS DU CLIENT --------------------------------------------------- -- Formulaire court de la garantie (nom, téléphone, numéro Wave). Modifiable tant que le prêt n'est pas payé. create or replace function public.guarantee_declare(p_request uuid, p_name text, p_phone text, p_wave text) returns void language plpgsql security definer set search_path = public as $$ begin if length(trim(coalesce(p_name, ''))) < 2 then raise exception 'Indiquez votre nom.'; end if; if length(regexp_replace(coalesce(p_phone, ''), '\D', '', 'g')) < 9 then raise exception 'Téléphone invalide.'; end if; if length(regexp_replace(coalesce(p_wave, ''), '\D', '', 'g')) < 9 then raise exception 'Numéro Wave invalide.'; end if; update public.requests set guarantee_name = trim(p_name), guarantee_phone = trim(p_phone), wave_number = trim(p_wave), guarantee_declared_at = coalesce(guarantee_declared_at, now()) where id = p_request and client_id = auth.uid() and status in ('pending', 'guarantee_received'); if not found then raise exception 'Cette demande ne peut plus être modifiée.'; end if; end $$; -- Le client a renvoyé l'argent sur le Wave de Nexus. create or replace function public.repayment_request(p_request uuid, p_amount int, p_ref text) returns void language plpgsql security definer set search_path = public as $$ begin if coalesce(p_amount, 0) <= 0 then raise exception 'Montant invalide.'; end if; if length(trim(coalesce(p_ref, ''))) < 4 then raise exception 'Indiquez la référence de la transaction Wave.'; end if; update public.requests set status = 'repayment_requested', repayment_amount = p_amount, repayment_ref = trim(p_ref), repayment_requested_at = now(), admin_note = null where id = p_request and client_id = auth.uid() and status = 'paid'; if not found then raise exception 'Remboursement impossible pour cette demande.'; end if; end $$; -- Rattache le contrat généré (une seule fois, dans le dossier du client). create or replace function public.contract_attach(p_request uuid, p_path text) returns void language plpgsql security definer set search_path = public as $$ begin if p_path is null or split_part(p_path, '/', 1) <> auth.uid()::text then raise exception 'Chemin invalide.'; end if; update public.requests set pdf_url = p_path, pdf_filename = 'contrat.html' where id = p_request and client_id = auth.uid() and pdf_url is null; end $$; -- 5. ACTIONS DE L'ADMIN (transitions contrôlées) -------------------------- create or replace function public.admin_set_status(p_request uuid, p_status text, p_note text default null) returns void language plpgsql security definer set search_path = public as $$ declare r public.requests; begin if not public.is_admin() then raise exception 'Réservé à l''administrateur.'; end if; select * into r from public.requests where id = p_request for update; if r.id is null then raise exception 'Demande introuvable.'; end if; if (r.status, p_status) not in ( ('pending', 'guarantee_received'), ('pending', 'rejected'), ('guarantee_received', 'pending'), ('guarantee_received', 'paid'), ('guarantee_received', 'rejected'), ('paid', 'repayment_requested'), ('repayment_requested', 'done'), ('repayment_requested', 'paid'), ('rejected', 'pending')) then raise exception 'Passage de « % » à « % » impossible.', r.status, p_status; end if; if p_status = 'paid' and r.status = 'guarantee_received' and r.wave_number is null then raise exception 'Le client n''a pas encore donné son numéro Wave.'; end if; update public.requests set status = p_status, admin_note = p_note, rejection_reason = case when p_status = 'rejected' then coalesce(p_note, 'Non précisé') else null end, guarantee_received_at = case when p_status = 'guarantee_received' then now() when p_status = 'pending' then null else guarantee_received_at end, paid_at = case when p_status = 'paid' and r.status = 'guarantee_received' then now() else paid_at end, paid_amount = case when p_status = 'paid' and r.status = 'guarantee_received' then r.montant else paid_amount end, repayment_requested_at= case when p_status = 'repayment_requested' then coalesce(repayment_requested_at, now()) else repayment_requested_at end, repaid_at = case when p_status = 'done' then now() else repaid_at end where id = p_request; end $$; revoke all on function public.guarantee_declare(uuid, text, text, text) from public, anon; revoke all on function public.repayment_request(uuid, int, text) from public, anon; revoke all on function public.contract_attach(uuid, text) from public, anon; grant execute on function public.contract_attach(uuid, text) to authenticated; grant execute on function public.admin_set_status(uuid, text, text) to authenticated; -- 6.
